cmake_minimum_required(VERSION 3.22)
project(Nexis VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(CTest)
include(GNUInstallDirs)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Network DBus)
find_package(LayerShellQt 6.7 QUIET)
add_library(nexis-core modules/agents/Events.cpp modules/media/Media.cpp)
target_include_directories(nexis-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(nexis-core PUBLIC Qt6::Core Qt6::Gui Qt6::Network Qt6::DBus)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 add_executable(test-core tests/unit/test_core.cpp)
 target_link_libraries(test-core PRIVATE nexis-core Qt6::Test)
 add_test(NAME core COMMAND test-core)
endif()
add_library(nexis-desktop services/core/Jobs.cpp services/core/Compositor.cpp services/core/Settings.cpp services/core/Collector.cpp services/core/Backend.cpp modules/system/System.cpp)
target_link_libraries(nexis-desktop PUBLIC nexis-core Qt6::Quick Qt6::QuickControls2)
add_executable(nexis apps/island/main.cpp apps/island/resources.qrc)
target_link_libraries(nexis PRIVATE nexis-desktop)
target_compile_definitions(nexis PRIVATE NEXIS_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(TARGET LayerShellQt::Interface)
 target_link_libraries(nexis PRIVATE LayerShellQt::Interface)
 get_filename_component(NEXIS_LAYER_PREFIX "${LayerShellQt_DIR}/../../.." ABSOLUTE)
 target_include_directories(nexis SYSTEM PRIVATE "${NEXIS_LAYER_PREFIX}/include")
 target_compile_definitions(nexis PRIVATE HAVE_LAYER_SHELL)
endif()
add_executable(nexis-hook-bridge tools/nexis-hook-bridge/main.cpp)
target_link_libraries(nexis-hook-bridge PRIVATE nexis-core)
if(BUILD_TESTING)
 find_package(Python3 COMPONENTS Interpreter REQUIRED)
 add_test(NAME hooks COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration -p test_hooks.py)
 add_test(NAME gui-smoke COMMAND dbus-run-session -- $<TARGET_FILE:nexis> --demo --windowed --smoke-test)
 file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/test-runtime)
 file(CHMOD ${CMAKE_CURRENT_BINARY_DIR}/test-runtime PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE)
 set_tests_properties(gui-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;XDG_RUNTIME_DIR=${CMAKE_CURRENT_BINARY_DIR}/test-runtime")
endif()
install(TARGETS nexis nexis-hook-bridge RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS integrations/hooks.py scripts/doctor.sh scripts/kwin-fullscreen.js DESTINATION ${CMAKE_INSTALL_DATADIR}/nexis)
install(FILES packaging/desktop/org.nexis.Desktop.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/systemd-user/nexis.service DESTINATION ${CMAKE_INSTALL_DATADIR}/systemd/user)
if(BUILD_TESTING)
 add_executable(test-media tests/integration/test_media.cpp)
 target_link_libraries(test-media PRIVATE nexis-core Qt6::Test)
 add_test(NAME media-bus COMMAND dbus-run-session -- $<TARGET_FILE:test-media>)
 add_test(NAME transport COMMAND dbus-run-session -- ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration -p test_transport.py)
 set_tests_properties(transport PROPERTIES ENVIRONMENT "NEXIS_BUILD_DIR=${CMAKE_CURRENT_BINARY_DIR}" TIMEOUT 15)
endif()

if(BUILD_TESTING)
 add_executable(test-desktop tests/unit/test_desktop.cpp)
 target_link_libraries(test-desktop PRIVATE nexis-desktop Qt6::Test)
 add_test(NAME desktop COMMAND dbus-run-session -- $<TARGET_FILE:test-desktop>)
 set_tests_properties(desktop PROPERTIES ENVIRONMENT "XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-config")
endif()
if(BUILD_TESTING)
 add_executable(test-gui tests/gui/test_gui.cpp apps/island/resources.qrc)
 target_link_libraries(test-gui PRIVATE nexis-desktop Qt6::Test)
 add_test(NAME gui-interaction COMMAND dbus-run-session -- $<TARGET_FILE:test-gui>)
 set_tests_properties(gui-interaction PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/gui-test-config;XDG_RUNTIME_DIR=${CMAKE_CURRENT_BINARY_DIR}/test-runtime")
endif()

install(FILES packaging/desktop/org.nexis.Desktop.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
target_sources(nexis-desktop PRIVATE services/core/Presentation.cpp)
if(BUILD_TESTING)
 add_executable(test-presentation tests/unit/test_presentation.cpp)
 target_link_libraries(test-presentation PRIVATE nexis-desktop Qt6::Test)
 add_test(NAME presentation COMMAND test-presentation)
endif()
add_library(nexis-panel-client STATIC apps/panel-client/DesktopClient.cpp)
set_target_properties(nexis-panel-client PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(nexis-panel-client PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(nexis-panel-client PUBLIC Qt6::Core Qt6::DBus)
if(BUILD_TESTING)
 add_executable(test-panel-client tests/integration/test_panel_client.cpp)
 target_link_libraries(test-panel-client PRIVATE nexis-panel-client Qt6::Test)
 add_test(NAME panel-client COMMAND dbus-run-session -- $<TARGET_FILE:test-panel-client>)
endif()
set(NEXIS_PLASMOID_DIR "${CMAKE_CURRENT_BINARY_DIR}/plasmoid/org.nexis.panel")
foreach(panel_file metadata.json contents/ui/main.qml contents/ui/Nexis/Panel/qmldir)
 configure_file("apps/plasmoid/${panel_file}" "${NEXIS_PLASMOID_DIR}/${panel_file}" COPYONLY)
endforeach()
configure_file(ui/TuxedoCat.qml "${NEXIS_PLASMOID_DIR}/contents/ui/TuxedoCat.qml" COPYONLY)
configure_file(ui/CompactActivity.qml "${NEXIS_PLASMOID_DIR}/contents/ui/CompactActivity.qml" COPYONLY)
add_library(nexispanelplugin MODULE apps/panel-client/Plugin.cpp)
target_link_libraries(nexispanelplugin PRIVATE nexis-panel-client Qt6::Qml)
set_target_properties(nexispanelplugin PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${NEXIS_PLASMOID_DIR}/contents/ui/Nexis/Panel")
install(DIRECTORY "${NEXIS_PLASMOID_DIR}/" DESTINATION ${CMAKE_INSTALL_DATADIR}/plasma/plasmoids/org.nexis.panel PATTERN "libnexispanelplugin.so" EXCLUDE)
install(TARGETS nexispanelplugin LIBRARY DESTINATION ${CMAKE_INSTALL_DATADIR}/plasma/plasmoids/org.nexis.panel/contents/ui/Nexis/Panel)
install(FILES packaging/systemd-user/nexis-panel.service DESTINATION ${CMAKE_INSTALL_DATADIR}/systemd/user)
if(BUILD_TESTING)
 add_test(NAME panel-install COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration -p test_panel_install.py)
endif()
